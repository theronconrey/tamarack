# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE: these commands erase every listed drive.
set -euo pipefail

# 1. Partition each drive
parted --script '/dev/disk/by-id/ata-WDC odd' mklabel gpt
parted --script '/dev/disk/by-id/ata-WDC odd' mkpart tk-meta 1MiB 17MiB
parted --script '/dev/disk/by-id/ata-WDC odd' mkpart tk-l1 17MiB 3814696MiB set 2 raid on
partprobe '/dev/disk/by-id/ata-WDC odd'
parted --script /dev/nvme0n1 mklabel gpt
parted --script /dev/nvme0n1 mkpart tk-meta 1MiB 17MiB
parted --script /dev/nvme0n1 mkpart tk-l1 17MiB 3814696MiB set 2 raid on
partprobe /dev/nvme0n1

# 2. dm-integrity on every layer partition (journal mode is integritysetup's default; bitmap mode stays off)
integritysetup format --batch-mode --integrity crc32c '/dev/disk/by-id/ata-WDC odd-part2'
integritysetup open --integrity crc32c '/dev/disk/by-id/ata-WDC odd-part2' tk-d1-p2
integritysetup format --batch-mode --integrity crc32c /dev/nvme0n1p2
integritysetup open --integrity crc32c /dev/nvme0n1p2 tk-d2-p2

# 3. One md array per layer
mdadm --create /dev/md/tk-l1 --run --metadata=1.2 --data-offset=128M --level=5 --raid-devices=2 --force --consistency-policy=ppl /dev/mapper/tk-d1-p2 /dev/mapper/tk-d2-p2

# 4. LVM joins the layers into one volume
wipefs --all /dev/md/tk-l1
pvcreate /dev/md/tk-l1
vgcreate tamarack /dev/md/tk-l1
lvcreate --yes --extents 100%FREE --name data tamarack

# 5. Btrfs on top
mkfs.btrfs --label tamarack --metadata dup /dev/tamarack/data
systemctl enable --now srv-tamarack.mount tamarack-scrub.timer
btrfs subvolume create /srv/tamarack/shares
chcon system_u:object_r:tamarack_data_t:s0 /srv/tamarack/shares

# 6. Snapshots of the shares: 24 hourly, 7 daily, 4 weekly, 6 monthly
#    snapper keeps them in shares/.snapshots; the schedule can be changed or turned off later.
snapper --config tamarack create-config --template tamarack --fstype btrfs /srv/tamarack/shares
systemctl enable --now snapper-timeline.timer snapper-cleanup.timer

# 7. Write the layout to every tk-meta partition
#    Each drive carries the whole layout, so the box can be read from any of them.
tamarack-meta write '/dev/disk/by-id/ata-WDC odd-part1' '{"drives":[{"device":"/dev/disk/by-id/ata-WDC odd","size":4000000000000},{"device":"/dev/nvme0n1","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
tamarack-meta write /dev/nvme0n1p1 '{"drives":[{"device":"/dev/disk/by-id/ata-WDC odd","size":4000000000000},{"device":"/dev/nvme0n1","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
