# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE for /dev/sdb: it will be erased.
set -euo pipefail

# 1. Remove the failed /dev/sdb from its layers
mdadm /dev/md/tk-l1 --remove failed
mdadm /dev/md/tk-l1 --remove detached
integritysetup close tk-d2-p2 || :

# 2. Partition /dev/sdb
sgdisk --zap-all /dev/sdb
sgdisk -n 1:0:+16M -t 1:8300 -c 1:tk-meta /dev/sdb
sgdisk -n 2:0:+3814679M -t 2:fd00 -c 2:tk-l1 /dev/sdb
partprobe /dev/sdb

# 3. dm-integrity on the new partitions
integritysetup format --batch-mode --integrity crc32c /dev/sdb2
integritysetup open --integrity crc32c /dev/sdb2 tk-d2-p2

# 4. Layer 1: rebuild onto the new drive
#    mdadm rebuilds this slice from the other drives. The layer is unprotected until it finishes.
mdadm /dev/md/tk-l1 --add /dev/mapper/tk-d2-p2
mdadm --wait /dev/md/tk-l1
