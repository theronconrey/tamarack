# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE for /dev/sdd: it will be erased.
set -euo pipefail

# 1. Check every layer is healthy
#    Stops here if an array is degraded: a second problem must be fixed first.
mdadm --detail --test /dev/md/tk-l1

# 2. Partition /dev/sdd
sgdisk --zap-all /dev/sdd
sgdisk -n 1:0:+16M -t 1:8300 -c 1:tk-meta /dev/sdd
sgdisk -n 2:0:+3814679M -t 2:fd00 -c 2:tk-l1 /dev/sdd
sgdisk -n 3:0:+3814697M -t 3:fd00 -c 3:tk-l2 /dev/sdd
partprobe /dev/sdd

# 3. dm-integrity on the new partitions
integritysetup format --batch-mode --integrity crc32c /dev/sdd2
integritysetup open --integrity crc32c /dev/sdd2 tk-d4-p2
integritysetup format --batch-mode --integrity crc32c /dev/sdd3
integritysetup open --integrity crc32c /dev/sdd3 tk-d4-p3

# 4. Layer 1: widen single parity to 4 drives
#    PPL is paused for the reshape and switched back on afterwards. The data stays online and protected; mdadm resumes from its checkpoint after a power cut.
mdadm --grow /dev/md/tk-l1 --consistency-policy=resync
mdadm /dev/md/tk-l1 --add /dev/mapper/tk-d4-p2
mdadm --grow /dev/md/tk-l1 --raid-devices=4
mdadm --wait /dev/md/tk-l1
mdadm --grow /dev/md/tk-l1 --consistency-policy=ppl

# 5. Add partitions in free space on /dev/sdc
#    Existing partitions are untouched.
sgdisk -n 3:0:+3814697M -t 3:fd00 -c 3:tk-l2 /dev/sdc
partprobe /dev/sdc
integritysetup format --batch-mode --integrity crc32c /dev/sdc3
integritysetup open --integrity crc32c /dev/sdc3 tk-d3-p3

# 6. Create layer 2 from leftover space
mdadm --create /dev/md/tk-l2 --run --metadata=1.2 --level=1 --raid-devices=2 --bitmap=internal /dev/mapper/tk-d3-p3 /dev/mapper/tk-d4-p3

# 7. Grow the volume and file system
pvresize /dev/md/tk-l1
pvcreate /dev/md/tk-l2
vgextend tamarack /dev/md/tk-l2
lvextend --extents +100%FREE tamarack/data
btrfs filesystem resize max /srv/tamarack
