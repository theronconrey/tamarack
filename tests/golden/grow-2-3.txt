# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE for /dev/sdc: it will be erased.
set -euo pipefail

# 1. Check every layer is healthy
#    Waits for any resync still running (a new array's first one, say), then stops here if an array is degraded: a second problem must be fixed first.
mdadm --wait /dev/md/tk-l1 || :
mdadm --detail --test /dev/md/tk-l1

# 2. Partition /dev/sdc
parted --script /dev/sdc mklabel gpt
parted --script /dev/sdc mkpart tk-meta 1MiB 17MiB
parted --script /dev/sdc mkpart tk-l1 17MiB 3814696MiB set 2 raid on
partprobe /dev/sdc

# 3. dm-integrity on the new partitions
integritysetup format --batch-mode --integrity crc32c /dev/sdc2
integritysetup open --integrity crc32c /dev/sdc2 tk-d3-p2

# 4. Layer 1: widen single parity to 3 drives
#    md cannot reshape with its partial parity log on, so the log is off for the reshape and back on afterwards. During the reshape a power cut while a drive is also missing can damage data, so keep the power on; mdadm resumes from its checkpoint after a power cut. Large layers take hours to days.
mdadm /dev/md/tk-l1 --add /dev/mapper/tk-d3-p2
mdadm --grow /dev/md/tk-l1 --consistency-policy=resync
mdadm --grow /dev/md/tk-l1 --raid-devices=3
mdadm --wait /dev/md/tk-l1 || :
mdadm --grow /dev/md/tk-l1 --consistency-policy=ppl

# 5. Grow the volume and file system
pvresize /dev/md/tk-l1
lvextend --extents +100%FREE tamarack/data
btrfs filesystem resize max /srv/tamarack

# 6. Write the new layout to every tk-meta partition
#    Each drive carries the whole layout, so the box can be read from any of them.
tamarack-meta write /dev/sda1 '{"drives":[{"device":"/dev/sda","size":4000000000000},{"device":"/dev/sdb","size":4000000000000},{"device":"/dev/sdc","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2},{"drive":2,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
tamarack-meta write /dev/sdb1 '{"drives":[{"device":"/dev/sda","size":4000000000000},{"device":"/dev/sdb","size":4000000000000},{"device":"/dev/sdc","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2},{"drive":2,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
tamarack-meta write /dev/sdc1 '{"drives":[{"device":"/dev/sda","size":4000000000000},{"device":"/dev/sdb","size":4000000000000},{"device":"/dev/sdc","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2},{"drive":2,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
