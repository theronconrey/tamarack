# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE: these commands erase every listed drive.
set -euo pipefail

# 1. Partition each drive
parted --script /dev/sda mklabel gpt
parted --script /dev/sda mkpart tk-meta 1MiB 17MiB
parted --script /dev/sda mkpart tk-l1a 17MiB 1907356MiB set 2 raid on
parted --script /dev/sda mkpart tk-l1b 1907356MiB 3814695MiB set 3 raid on
partprobe /dev/sda

# 2. dm-integrity on every layer partition (journal mode is integritysetup's default; bitmap mode stays off)
integritysetup format --batch-mode --integrity crc32c /dev/sda2
integritysetup open --integrity crc32c /dev/sda2 tk-d1-p2
integritysetup format --batch-mode --integrity crc32c /dev/sda3
integritysetup open --integrity crc32c /dev/sda3 tk-d1-p3

# 3. One md array per layer
mdadm --create /dev/md/tk-l1 --run --metadata=1.2 --level=1 --raid-devices=2 --bitmap=internal /dev/mapper/tk-d1-p2 /dev/mapper/tk-d1-p3

# 4. LVM joins the layers into one volume
pvcreate /dev/md/tk-l1
vgcreate tamarack /dev/md/tk-l1
lvcreate --yes --extents 100%FREE --name data tamarack

# 5. Btrfs on top
mkfs.btrfs --label tamarack --metadata dup /dev/tamarack/data
systemctl enable --now srv-tamarack.mount
btrfs subvolume create /srv/tamarack/shares
btrfs subvolume create /srv/tamarack/snapshots

# 6. Write the layout to every tk-meta partition
#    Each drive carries the whole layout, so the box can be read from any of them.
tamarack-meta write /dev/sda1 '{"drives":[{"device":"/dev/sda","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid1","members":[{"drive":0,"partition":2},{"drive":0,"partition":3}],"slice_bytes":1999989899264}],"version":1}'
