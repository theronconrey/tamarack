# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE for /dev/sdd: it will be erased.
set -euo pipefail

# 1. Check every layer is healthy
#    Stops here if an array is degraded: a second problem must be fixed first.
mdadm --detail --test /dev/md/tk-l1

# 2. Partition /dev/sdd
parted --script /dev/sdd mklabel gpt
parted --script /dev/sdd mkpart tk-meta 1MiB 17MiB
parted --script /dev/sdd mkpart tk-l1 17MiB 3814696MiB set 2 raid on
partprobe /dev/sdd

# 3. dm-integrity on the new partitions
integritysetup format --batch-mode --integrity crc32c /dev/sdd2
integritysetup open --integrity crc32c /dev/sdd2 tk-d2-p2-new

# 4. Layer 1: copy onto the new drive
#    The old copy stays active until the new one is complete, so protection never drops.
mdadm /dev/md/tk-l1 --add /dev/mapper/tk-d2-p2-new
mdadm /dev/md/tk-l1 --replace /dev/mapper/tk-d2-p2 --with /dev/mapper/tk-d2-p2-new
mdadm --wait /dev/md/tk-l1
mdadm /dev/md/tk-l1 --remove /dev/mapper/tk-d2-p2

# 5. Retire the old drive
#    Closes the old drive's mappings and gives the new ones their permanent names. The old drive can be removed after this step.
integritysetup close tk-d2-p2
dmsetup rename tk-d2-p2-new tk-d2-p2

# 6. Write the new layout to every tk-meta partition
#    Each drive carries the whole layout, so the box can be read from any of them.
tamarack-meta write /dev/sda1 '{"drives":[{"device":"/dev/sda","size":4000000000000},{"device":"/dev/sdd","size":4000000000000},{"device":"/dev/sdc","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2},{"drive":2,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
tamarack-meta write /dev/sdd1 '{"drives":[{"device":"/dev/sda","size":4000000000000},{"device":"/dev/sdd","size":4000000000000},{"device":"/dev/sdc","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2},{"drive":2,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
tamarack-meta write /dev/sdc1 '{"drives":[{"device":"/dev/sda","size":4000000000000},{"device":"/dev/sdd","size":4000000000000},{"device":"/dev/sdc","size":4000000000000}],"format":"tamarack-layout","layers":[{"index":1,"level":"raid5","members":[{"drive":0,"partition":2},{"drive":1,"partition":2},{"drive":2,"partition":2}],"slice_bytes":3999980847104}],"version":1}'
