# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE: these commands erase every listed drive.
set -euo pipefail

# 1. Partition each drive
parted --script /dev/sda mklabel gpt
parted --script /dev/sda mkpart tk-meta 1MiB 17MiB
parted --script /dev/sda mkpart tk-l1 17MiB 3814696MiB set 2 raid on
partprobe /dev/sda
parted --script /dev/sdb mklabel gpt
parted --script /dev/sdb mkpart tk-meta 1MiB 17MiB
parted --script /dev/sdb mkpart tk-l1 17MiB 3814696MiB set 2 raid on
partprobe /dev/sdb
parted --script /dev/sdc mklabel gpt
parted --script /dev/sdc mkpart tk-meta 1MiB 17MiB
parted --script /dev/sdc mkpart tk-l1 17MiB 3814696MiB set 2 raid on
parted --script /dev/sdc mkpart tk-l2 3814696MiB 7629393MiB set 3 raid on
partprobe /dev/sdc
parted --script /dev/sdd mklabel gpt
parted --script /dev/sdd mkpart tk-meta 1MiB 17MiB
parted --script /dev/sdd mkpart tk-l1 17MiB 3814696MiB set 2 raid on
parted --script /dev/sdd mkpart tk-l2 3814696MiB 7629393MiB set 3 raid on
partprobe /dev/sdd

# 2. dm-integrity on every layer partition (journal mode is integritysetup's default; bitmap mode stays off)
integritysetup format --batch-mode --integrity crc32c /dev/sda2
integritysetup open --integrity crc32c /dev/sda2 tk-d1-p2
integritysetup format --batch-mode --integrity crc32c /dev/sdb2
integritysetup open --integrity crc32c /dev/sdb2 tk-d2-p2
integritysetup format --batch-mode --integrity crc32c /dev/sdc2
integritysetup open --integrity crc32c /dev/sdc2 tk-d3-p2
integritysetup format --batch-mode --integrity crc32c /dev/sdd2
integritysetup open --integrity crc32c /dev/sdd2 tk-d4-p2
integritysetup format --batch-mode --integrity crc32c /dev/sdc3
integritysetup open --integrity crc32c /dev/sdc3 tk-d3-p3
integritysetup format --batch-mode --integrity crc32c /dev/sdd3
integritysetup open --integrity crc32c /dev/sdd3 tk-d4-p3

# 3. One md array per layer
mdadm --create /dev/md/tk-l1 --run --metadata=1.2 --homehost=any --data-offset=128M --level=5 --raid-devices=4 --force --consistency-policy=ppl /dev/mapper/tk-d1-p2 /dev/mapper/tk-d2-p2 /dev/mapper/tk-d3-p2 /dev/mapper/tk-d4-p2
mdadm --create /dev/md/tk-l2 --run --metadata=1.2 --homehost=any --data-offset=128M --level=5 --raid-devices=2 --force --consistency-policy=ppl /dev/mapper/tk-d3-p3 /dev/mapper/tk-d4-p3

# 4. LVM joins the layers into one volume
wipefs --all /dev/md/tk-l1
wipefs --all /dev/md/tk-l2
pvcreate /dev/md/tk-l1 /dev/md/tk-l2
vgcreate tamarack /dev/md/tk-l1 /dev/md/tk-l2
lvcreate --yes --extents 100%FREE --name data tamarack

# 5. Btrfs on top
mkfs.btrfs --label tamarack --metadata dup /dev/tamarack/data
systemctl enable --now srv-tamarack.mount tamarack-scrub.timer
btrfs subvolume create /srv/tamarack/shares
chcon system_u:object_r:tamarack_data_t:s0 /srv/tamarack/shares

# 6. Snapshots of the shares: 24 hourly, 7 daily, 4 weekly, 6 monthly
#    snapper keeps them in shares/.snapshots; the schedule can be changed or turned off later.
snapper --config tamarack create-config --template tamarack --fstype btrfs /srv/tamarack/shares
systemctl enable --now snapper-timeline.timer snapper-cleanup.timer
