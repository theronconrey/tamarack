# tamarack dry run: nothing below has been executed.
# DESTRUCTIVE for /dev/sdf: it will be erased.
set -euo pipefail

# 1. Check every layer is healthy
#    Stops here if an array is degraded: a second problem must be fixed first.
mdadm --detail --test /dev/md/tk-l1

# 2. Partition /dev/sdf
sgdisk --zap-all /dev/sdf
sgdisk -n 1:0:+16M -t 1:8300 -c 1:tk-meta /dev/sdf
sgdisk -n 2:0:+3814679M -t 2:fd00 -c 2:tk-l1 /dev/sdf
sgdisk -n 3:0:+3814697M -t 3:fd00 -c 3:tk-l2 /dev/sdf
partprobe /dev/sdf

# 3. dm-integrity on the new partitions
integritysetup format --batch-mode --integrity crc32c /dev/sdf2
integritysetup open --integrity crc32c /dev/sdf2 tk-d2-p2-new
integritysetup format --batch-mode --integrity crc32c /dev/sdf3
integritysetup open --integrity crc32c /dev/sdf3 tk-d2-p3

# 4. Layer 1: copy onto the new drive
#    The old copy stays active until the new one is complete, so protection never drops.
mdadm /dev/md/tk-l1 --add /dev/mapper/tk-d2-p2-new
mdadm /dev/md/tk-l1 --replace /dev/mapper/tk-d2-p2 --with /dev/mapper/tk-d2-p2-new
mdadm --wait /dev/md/tk-l1
mdadm /dev/md/tk-l1 --remove /dev/mapper/tk-d2-p2

# 5. Retire the old drive
#    Closes the old drive's mappings and gives the new ones their permanent names. The old drive can be removed after this step.
integritysetup close tk-d2-p2
dmsetup rename tk-d2-p2-new tk-d2-p2

# 6. Add partitions in free space on /dev/sde
#    Existing partitions are untouched.
sgdisk -n 3:0:+3814697M -t 3:fd00 -c 3:tk-l2 /dev/sde
partprobe /dev/sde
integritysetup format --batch-mode --integrity crc32c /dev/sde3
integritysetup open --integrity crc32c /dev/sde3 tk-d1-p3

# 7. Create layer 2 from leftover space
mdadm --create /dev/md/tk-l2 --run --metadata=1.2 --level=1 --raid-devices=2 --bitmap=internal /dev/mapper/tk-d1-p3 /dev/mapper/tk-d2-p3

# 8. Grow the volume and file system
pvcreate /dev/md/tk-l2
vgextend tamarack /dev/md/tk-l2
lvextend --extents +100%FREE tamarack/data
btrfs filesystem resize max /srv/tamarack
